cmake_minimum_required(VERSION 3.25)
project(ControllerFocus VERSION 1.0.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(BUILD_TESTING)
    add_executable(focus_gate_tests tests/gate.cpp)
    add_test(NAME focus_transitions COMMAND focus_gate_tests)
endif()
option(FOCUS_GUARD_TESTS_ONLY "Build standalone tests without Geode" OFF)
if(FOCUS_GUARD_TESTS_ONLY)
    return()
endif()
if(NOT WIN32)
    message(FATAL_ERROR "Controller Focus currently supports Windows only")
endif()
add_library(ControllerFocus SHARED src/main.cpp)
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
target_link_libraries(ControllerFocus user32 xinput9_1_0)
setup_geode_mod(ControllerFocus DONT_INSTALL)
